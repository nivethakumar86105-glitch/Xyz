10. Text Preprocessing
=======================================================================================================
python
------
import re

def clean_text(text):
    text = text.lower()

    text = re.sub(r"[^a-zA-Z\s]", " ", text)

    text = re.sub(r"\s+", " ", text)

    return text.strip()
=======================================================================================================
11. Train/Test Split
=======================================================================================================
python
------
from sklearn.model_selection import train_test_split

X = df["text"].apply(clean_text)
y = df["label"]

X_train, X_test, y_train, y_test = train_test_split(X,y,test_size=0.25,random_state=42,stratify=y)
=======================================================================================================
12. TF-IDF[Term Frequency-Inverse Document Frequency]
=======================================================================================================
python
------
from sklearn.feature_extraction.text import TfidfVectorizer

vectorizer = TfidfVectorizer(stop_words="english",ngram_range=(1, 2))

X_train_tfidf = vectorizer.fit_transform(X_train)

X_test_tfidf = vectorizer.transform(X_test)


Remember
--------
Training -> fit_transform()
Testing  -> transform()
=======================================================================================================
13. Train ML Model
=======================================================================================================
python
------
from sklearn.linear_model import LogisticRegression

model = LogisticRegression(max_iter=1000)

model.fit(X_train_tfidf,y_train)

print("Training completed")

Output
------
Training completed

17. Complete Tkinter Application(Desktop)
=======================================================================================================
tkinter_app.py
--------------
python
------
import tkinter as tk
from tkinter import messagebox

import joblib


# Load trained model
model = joblib.load("phishing_model.pkl")

vectorizer = joblib.load("tfidf_vectorizer.pkl")


# -------------------------------
# Prediction Function
# -------------------------------

def check_email():

    email = email_box.get("1.0",tk.END).strip()

    if not email:

        messagebox.showwarning("Input Required","Please enter email text.")

        return


    # Convert email into TF-IDF
    email_vector = vectorizer.transform([email])


    # Prediction
    prediction = model.predict(email_vector)[0]


    # Display result
    if prediction == "phishing":

        result_label.config(text="PHISHING EMAIL")

    else:

        result_label.config(text="? EMAIL APPEARS SAFE")


# -------------------------------
# Clear Function
# -------------------------------

def clear_email():

    email_box.delete("1.0",tk.END)

    result_label.config(text="")


# -------------------------------
# Main Window
# -------------------------------

root = tk.Tk()

root.title("AI Phishing Email Detector")

root.geometry("650x500")


# -------------------------------
# Heading
# -------------------------------

title_label = tk.Label(root,text="AI-Powered Phishing Email Detector",font=("Arial", 20, "bold"))

title_label.pack(pady=20)


# -------------------------------
# Input Label
# -------------------------------

input_label = tk.Label(root,text="Enter Email Text:",font=("Arial", 12))

input_label.pack()


# -------------------------------
# Email Text Box
# -------------------------------

email_box = tk.Text(root,height=10,width=70)

email_box.pack(pady=10)


# -------------------------------
# Check Button
# -------------------------------

check_button = tk.Button(root, text="Check Email", command=check_email, font=("Arial", 12))

check_button.pack(pady=5)


# -------------------------------
# Clear Button
# -------------------------------

clear_button = tk.Button(root, text="Clear", command=clear_email, font=("Arial", 12))

clear_button.pack(pady=5)


# -------------------------------
# Result
# -------------------------------

result_label = tk.Label(root, text="", font=("Arial", 18, "bold"))

result_label.pack(pady=25)


# -------------------------------
# Start GUI
# -------------------------------

root.mainloop()

Streamlit gives a web-based GUI.


Python ML Model
       ?

Desktop      Web
GUI          UI


app.py
------
python
------
import joblib
import streamlit as st


model = joblib.load("phishing_model.pkl")

vectorizer = joblib.load("tfidf_vectorizer.pkl")


st.title("AI Phishing Email Detector")

email = st.text_area("Enter Email Text")


if st.button("Check Email"):

    if not email.strip():

        st.warning("Please enter email text.")

    else:

        email_vector = vectorizer.transform([email])

        prediction = model.predict(email_vector)[0]

        if prediction == "phishing":

            st.error("PHISHING EMAIL")

        else:

            st.success("EMAIL APPEARS SAFE")

Run:
----
streamlit run app.py
